var SHEET_ID = "1XqW6w9QZErNl08hNKTferQzOlU73U7kTf_2FinPkyAY"; // Handles GET requests (Balance, Spend, Register) function doGet(e) { var userId = e.parameter.userId.toString(); var username = e.parameter.username || "User"; var action = e.parameter.action; var ss = SpreadsheetApp.openById(SHEET_ID); var sheet = ss.getSheets()[0]; var data = sheet.getDataRange().getValues(); var rowIndex = -1; // Find user row for (var i = 1; i < data.length; i++) { if (data[i][0].toString() === userId) { rowIndex = i + 1; break; } } // 1. REGISTER ON START if (action === "start") { if (rowIndex === -1) { sheet.appendRow([userId, username, 0, 0, 0]); // ID, Name, Earned, Spent, Balance return ContentService.createTextOutput("Registered"); } return ContentService.createTextOutput("Welcome back"); } // 2. GET STATS if (action === "get_stats") { if (rowIndex === -1) return ContentService.createTextOutput("0,0,0"); var row = data[rowIndex - 1]; // Returns "Earned,Spent,Balance" return ContentService.createTextOutput(row[2] + "," + row[3] + "," + row[4]); } // 3. SPEND POINTS if (action === "spend") { if (rowIndex !== -1) { var currentEarned = Number(data[rowIndex - 1][2]); sheet.getRange(rowIndex, 4).setValue(currentEarned); // Total Spent = Total Earned sheet.getRange(rowIndex, 5).setValue(0); // Balance = 0 return ContentService.createTextOutput("Spent"); } } } // Handles POST requests (Adding points from the Web App) function doPost(e) { var userId = e.parameter.userId.toString(); var ss = SpreadsheetApp.openById(SHEET_ID); var sheet = ss.getSheets()[0]; var data = sheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { if (data[i][0].toString() === userId) { var currentEarned = Number(data[i][2]); var currentSpent = Number(data[i][3]); var newEarned = currentEarned + 10; sheet.getRange(i + 1, 3).setValue(newEarned); // Update Earned sheet.getRange(i + 1, 5).setValue(newEarned - currentSpent); // Update Balance break; } } return ContentService.createTextOutput("Success"); }
